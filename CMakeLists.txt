cmake_minimum_required(VERSION 3.28)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_BINARY_DIR)
    message(FATAL_ERROR "In-source builds are unsupported. Use cmake --preset dev or headless.")
endif()
project(siggen VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(SIGGEN_BUILD_APP "Build the signal generator executable" ON)
option(SIGGEN_BUILD_GUI "Include the optional ImGui interface" ON)
option(SIGGEN_BUILD_TESTS "Build DSP tests" OFF)
option(SIGGEN_ENABLE_WARNINGS "Enable project compiler warnings" ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
include(cmake/ProjectOptions.cmake)
function(siggen_require_submodule name file)
    if(NOT EXISTS "${PROJECT_SOURCE_DIR}/vendor/${name}/${file}")
        message(FATAL_ERROR "Missing required ${name} submodule. Run: git submodule update --init --recursive")
    endif()
endfunction()
find_program(CLANG_FORMAT clang-format)
if(CLANG_FORMAT)
    file(GLOB_RECURSE PROJECT_SOURCES application/*.cpp application/*.h library/*.cpp library/*.h tests/*.cpp)
    add_custom_target(format COMMAND ${CLANG_FORMAT} -i ${PROJECT_SOURCES} VERBATIM)
endif()
if(SIGGEN_BUILD_APP OR SIGGEN_BUILD_GUI OR SIGGEN_BUILD_TESTS)
    find_package(Threads REQUIRED)
endif()
add_subdirectory(library)
if(SIGGEN_BUILD_GUI)
    siggen_require_submodule(glfw CMakeLists.txt)
    siggen_require_submodule(imgui imgui.cpp)
    siggen_require_submodule(implot implot.cpp)
    siggen_require_submodule(stb stb_image.h)
endif()
if(SIGGEN_BUILD_APP OR SIGGEN_BUILD_GUI)
    siggen_require_submodule(spdlog CMakeLists.txt)
    siggen_require_submodule(cli11 CMakeLists.txt)
    add_subdirectory(vendor)
endif()
if(SIGGEN_BUILD_GUI)
    add_library(siggen_ui STATIC application/signal_generator.cpp)
    target_include_directories(siggen_ui PUBLIC application)
    target_include_directories(siggen_ui SYSTEM PRIVATE vendor/stb)
    target_link_libraries(siggen_ui PUBLIC imgui implot signalgenerator Threads::Threads)
    target_link_libraries(siggen_ui PRIVATE spdlog::spdlog)
    siggen_target_options(siggen_ui)
endif()
if(SIGGEN_BUILD_APP)
    add_executable(siggen application/main.cpp application/cli_config.cpp)
    target_link_libraries(siggen PRIVATE signalgenerator spdlog::spdlog CLI11::CLI11)
    target_compile_definitions(siggen PRIVATE SIGGEN_VERSION="${PROJECT_VERSION}"
        SIGGEN_HAS_GUI=$<BOOL:${SIGGEN_BUILD_GUI}>)
    siggen_target_options(siggen)
    if(SIGGEN_BUILD_GUI)
        target_sources(siggen PRIVATE application/gui.cpp application/app.cpp
            application/imgui_layer.cpp application/window_manager.cpp)
        if(WIN32)
            target_sources(siggen PRIVATE resources.rc)
        endif()
        target_include_directories(siggen SYSTEM PRIVATE vendor/stb)
        target_link_libraries(siggen PRIVATE siggen_ui)
        if(EXISTS "${CMAKE_SOURCE_DIR}/assets")
            add_custom_command(TARGET siggen POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory
                "${CMAKE_SOURCE_DIR}/assets" "$<TARGET_FILE_DIR:siggen>/assets")
        endif()
    endif()
endif()
if(SIGGEN_BUILD_TESTS)
    enable_testing()
    siggen_require_submodule(googletest CMakeLists.txt)
    set(BUILD_GMOCK OFF CACHE BOOL "Build GoogleMock")
    set(INSTALL_GTEST OFF CACHE BOOL "Install GoogleTest")
    if(MSVC)
        set(gtest_force_shared_crt ON CACHE BOOL "Use the shared MSVC runtime")
    endif()
    add_subdirectory(vendor/googletest EXCLUDE_FROM_ALL)
    add_subdirectory(tests)
endif()
